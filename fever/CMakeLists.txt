cmake_minimum_required(VERSION 3.22)
project(Fever VERSION 1.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX)
if(UNIX AND NOT APPLE)
 add_compile_definitions(JUCE_USE_XINPUT=0)
endif()
if(CMAKE_CROSSCOMPILING)
 set(FEVER_MANIFEST FALSE)
else()
 set(FEVER_MANIFEST TRUE)
endif()
add_subdirectory(ext/JUCE EXCLUDE_FROM_ALL)
add_subdirectory(ext/clap-juce-extensions EXCLUDE_FROM_ALL)
juce_add_plugin(Fever COMPANY_NAME "CMPLXTY" PRODUCT_NAME "CMPLXTY FEVER"
 BUNDLE_ID com.cmplxty.fever PLUGIN_MANUFACTURER_CODE Cmpl PLUGIN_CODE Fvr1
 FORMATS VST3 Standalone IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 VST3_CATEGORIES Fx Modulation COPY_PLUGIN_AFTER_BUILD FALSE
 NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE VST3_AUTO_MANIFEST ${FEVER_MANIFEST})
clap_juce_extensions_plugin(TARGET Fever CLAP_ID "com.cmplxty.fever"
 CLAP_FEATURES audio-effect pitch-shifter delay reverb filter granular stereo mono)
include(cmake/win-version-info.cmake)
cmplxty_win_version_info(Fever_CLAP "CMPLXTY FEVER" "CMPLXTY")
juce_add_binary_data(FeverAssets SOURCES assets/fonts/Barlow-Regular.ttf assets/fonts/Barlow-SemiBold.ttf)
set_target_properties(FeverAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(FEVER_SOURCES src/PluginProcessor.cpp src/PluginEditor.cpp)
function(fever_config target)
 target_include_directories(${target} PRIVATE src)
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(${target} PRIVATE FeverAssets juce::juce_audio_utils juce::juce_recommended_config_flags ${CMAKE_DL_LIBS})
endfunction()
target_sources(Fever PRIVATE ${FEVER_SOURCES})
fever_config(Fever)
if(NOT CMAKE_CROSSCOMPILING)
 juce_add_console_app(FeverTest PRODUCT_NAME "FeverTest")
 target_sources(FeverTest PRIVATE test/FeverTest.cpp ${FEVER_SOURCES})
 fever_config(FeverTest)
endif()
