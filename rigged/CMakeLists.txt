cmake_minimum_required(VERSION 3.22)
project(Rigged VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX)
if(UNIX AND NOT APPLE)
 add_compile_definitions(JUCE_USE_XINPUT=0)
endif()
if(CMAKE_CROSSCOMPILING)
 set(RIGGED_MANIFEST FALSE)
else()
 set(RIGGED_MANIFEST TRUE)
endif()
add_subdirectory(ext/JUCE EXCLUDE_FROM_ALL)
add_subdirectory(ext/clap-juce-extensions EXCLUDE_FROM_ALL)
juce_add_plugin(Rigged COMPANY_NAME "CMPLXTY" PRODUCT_NAME "CMPLXTY RIGGED"
 BUNDLE_ID com.cmplxty.rigged PLUGIN_MANUFACTURER_CODE Cmpl PLUGIN_CODE Rgd1
 FORMATS VST3 Standalone IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 VST3_CATEGORIES Fx Modulation COPY_PLUGIN_AFTER_BUILD FALSE
 NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE VST3_AUTO_MANIFEST ${RIGGED_MANIFEST})
clap_juce_extensions_plugin(TARGET Rigged CLAP_ID "com.cmplxty.rigged"
 CLAP_FEATURES audio-effect filter distortion delay reverb modulator stereo mono)
include(cmake/win-version-info.cmake)
cmplxty_win_version_info(Rigged_CLAP "CMPLXTY RIGGED" "CMPLXTY")
juce_add_binary_data(RiggedAssets SOURCES assets/fonts/Barlow-Regular.ttf assets/fonts/Barlow-SemiBold.ttf)
set_target_properties(RiggedAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(RIGGED_SOURCES src/PluginProcessor.cpp src/PluginEditor.cpp)
function(rigged_config target)
 target_include_directories(${target} PRIVATE src)
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(${target} PRIVATE RiggedAssets juce::juce_audio_utils juce::juce_recommended_config_flags ${CMAKE_DL_LIBS})
endfunction()
target_sources(Rigged PRIVATE ${RIGGED_SOURCES})
rigged_config(Rigged)
if(NOT CMAKE_CROSSCOMPILING)
 juce_add_console_app(RiggedTest PRODUCT_NAME "RiggedTest")
 target_sources(RiggedTest PRIVATE test/RiggedTest.cpp ${RIGGED_SOURCES})
 rigged_config(RiggedTest)
endif()
