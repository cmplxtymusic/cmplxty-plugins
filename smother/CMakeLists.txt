cmake_minimum_required(VERSION 3.22)
project(Smother VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX)
if(UNIX AND NOT APPLE)
 add_compile_definitions(JUCE_USE_XINPUT=0)
endif()
if(CMAKE_CROSSCOMPILING)
 set(SMOTHER_MANIFEST FALSE)
else()
 set(SMOTHER_MANIFEST TRUE)
endif()
add_subdirectory(ext/JUCE EXCLUDE_FROM_ALL)
add_subdirectory(ext/clap-juce-extensions EXCLUDE_FROM_ALL)
juce_add_plugin(Smother COMPANY_NAME "CMPLXTY" PRODUCT_NAME "CMPLXTY SMOTHER"
 BUNDLE_ID com.cmplxty.smother PLUGIN_MANUFACTURER_CODE Cmpl PLUGIN_CODE Smt1
 FORMATS VST3 Standalone IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
 VST3_CATEGORIES Instrument Drum COPY_PLUGIN_AFTER_BUILD FALSE
 NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE VST3_AUTO_MANIFEST ${SMOTHER_MANIFEST})
clap_juce_extensions_plugin(TARGET Smother CLAP_ID "com.cmplxty.smother" CLAP_FEATURES instrument drum-machine stereo mono)
include(cmake/win-version-info.cmake)
cmplxty_win_version_info(Smother_CLAP "CMPLXTY SMOTHER" "CMPLXTY")
juce_add_binary_data(SmotherAssets SOURCES assets/fonts/Barlow-Regular.ttf assets/fonts/Barlow-SemiBold.ttf)
set_target_properties(SmotherAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(SMOTHER_SOURCES src/PluginProcessor.cpp src/PluginEditor.cpp)
function(smother_config target)
 target_include_directories(${target} PRIVATE src)
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(${target} PRIVATE SmotherAssets juce::juce_audio_utils juce::juce_recommended_config_flags ${CMAKE_DL_LIBS})
endfunction()
target_sources(Smother PRIVATE ${SMOTHER_SOURCES})
smother_config(Smother)
if(NOT CMAKE_CROSSCOMPILING)
 juce_add_console_app(SmotherTest PRODUCT_NAME "SmotherTest")
 target_sources(SmotherTest PRIVATE test/SmotherTest.cpp ${SMOTHER_SOURCES})
 smother_config(SmotherTest)
endif()
