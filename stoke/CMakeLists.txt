cmake_minimum_required(VERSION 3.22)
project(Stoke VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX)
if(UNIX AND NOT APPLE)
 add_compile_definitions(JUCE_USE_XINPUT=0)
endif()
if(CMAKE_CROSSCOMPILING)
 set(STOKE_MANIFEST FALSE)
else()
 set(STOKE_MANIFEST TRUE)
endif()
add_subdirectory(ext/JUCE EXCLUDE_FROM_ALL)
add_subdirectory(ext/clap-juce-extensions EXCLUDE_FROM_ALL)
juce_add_plugin(Stoke COMPANY_NAME "CMPLXTY" PRODUCT_NAME "CMPLXTY STOKE"
 BUNDLE_ID com.cmplxty.stoke PLUGIN_MANUFACTURER_CODE Cmpl PLUGIN_CODE Stk1
 FORMATS VST3 Standalone IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 VST3_CATEGORIES Fx EQ COPY_PLUGIN_AFTER_BUILD FALSE
 NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE VST3_AUTO_MANIFEST ${STOKE_MANIFEST})
clap_juce_extensions_plugin(TARGET Stoke CLAP_ID "com.cmplxty.stoke"
 CLAP_FEATURES audio-effect equalizer saturator filter stereo mono)
include(cmake/win-version-info.cmake)
cmplxty_win_version_info(Stoke_CLAP "CMPLXTY STOKE" "CMPLXTY")
juce_add_binary_data(StokeAssets SOURCES assets/fonts/Barlow-Regular.ttf assets/fonts/Barlow-SemiBold.ttf)
set_target_properties(StokeAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(STOKE_SOURCES src/PluginProcessor.cpp src/PluginEditor.cpp)
function(stoke_config target)
 target_include_directories(${target} PRIVATE src)
 target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(${target} PRIVATE StokeAssets juce::juce_audio_utils juce::juce_recommended_config_flags ${CMAKE_DL_LIBS})
endfunction()
target_sources(Stoke PRIVATE ${STOKE_SOURCES})
stoke_config(Stoke)
if(NOT CMAKE_CROSSCOMPILING)
 juce_add_console_app(StokeTest PRODUCT_NAME "StokeTest")
 target_sources(StokeTest PRIVATE test/StokeTest.cpp ${STOKE_SOURCES})
 stoke_config(StokeTest)
endif()
